\begin{frame}{Problem Formulation and The Biological Hierarchy}
\begin{columns}[onlytextwidth, t]
  \begin{column}{0.47\textwidth}
    \begin{block}{The Clinical Challenge: Placenta Histology}
      \parbox[t][4.9cm][t]{\linewidth}{
        \begin{itemize}\small
          \item The placenta is the vital organ governing fetal oxygenation and nutrition; pathological lesions cause \textbf{preeclampsia, fetal growth restriction (FGR), and stillbirth}.
          \item Digitized Whole Slide Images (WSIs) are \textbf{gigapixel scans} (billions of pixels) with massive cellular heterogeneity and specialist shortages.
          \item Patch-level CNNs fail because tissue semantics depend on \textbf{spatial cell arrangements} (similar cell types can form completely different tissue microstructures).
        \end{itemize}
      }
    \end{block}
  \end{column}
  \hfill
  \begin{column}{0.47\textwidth}
    \begin{block}{The HAPPY Biological Hierarchy}
      \parbox[t][4.9cm][t]{\linewidth}{
        \begin{itemize}\small
          \item \textbf{HAPPY}: \textit{Histology Analysis Pipeline.PY} (\textit{Nature Communications}, 2024).
          \item Implements an end-to-end 3-tier bottom-up hierarchy:
            \begin{enumerate}\footnotesize
              \item \textbf{Stage 1 (Locate)}: RetinaNet detects all nuclear bounding boxes and centroids on $1024\times 1024$ WSI tiles.
              \item \textbf{Stage 2 (Classify)}: ResNet-50 classifies $224\times 224$ cell crops into 11 lineages and extracts 64-d embeddings.
              \item \textbf{Stage 3 (Infer Structure)}: ClusterGCN builds spatial cell graphs to predict 8 functional tissue microenvironments.
            \end{enumerate}
        \end{itemize}
      }
    \end{block}
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Novelty: What We Did Apart from Replicating the Paper}
\begin{columns}[onlytextwidth, t]
  \begin{column}{0.47\textwidth}
    \begin{alertblock}{Base Paper Limitations Identified}
      \parbox[t][4.8cm][t]{\linewidth}{
        \begin{itemize}\footnotesize
          \item \textbf{Flat Vanilla Architecture}: Original Stage 2 used a plain ResNet-50 with unweighted cross-entropy and basic ReLU.
          \item \textbf{Severe Class Skew ($>14:1$)}: Abundant syncytiotrophoblasts starved rare clinical lineages of gradients:
            \begin{itemize}\scriptsize
              \item Maternal Decidua: only \textbf{54.5\% recall}.
              \item Hofbauer Macrophages: only \textbf{62.2\% recall}.
            \end{itemize}
          \item \textbf{Broken Graph Intersection}: Authors' structured array intersection broke on modern NumPy, retaining only 7 edges.
        \end{itemize}
      }
    \end{alertblock}
  \end{column}
  \hfill
  \begin{column}{0.47\textwidth}
    \begin{block}{Our Novel Contributions Beyond Replication}
      \parbox[t][4.8cm][t]{\linewidth}{
        \begin{itemize}\footnotesize
          \item \textbf{1. Residual Channel Attention (SE)}: Added a 1D Squeeze-and-Excitation block over 2,048-dim vectors ($\mathbf{\tilde{x}} = \mathbf{x} \odot \mathbf{s} + \mathbf{x}$) to boost stain-discriminative hyperchromatic channels.
          \item \textbf{2. Smooth Mish Activation}: Replaced dying ReLU with non-monotonic $\text{Mish}(x) = x \tanh(\ln(1+e^x))$ for continuous gradient flow.
          \item \textbf{3. Multi-Class Focal Loss ($\gamma = 2.0$)}: Dynamically downweights easy majority classes, focusing learning on rare cells.
          \item \textbf{4. 64-Bit Bit-Packed Graph Engine}: Developed \texttt{((u<<32)|v)} indexing, restoring all \textbf{5.08M edges} in seconds.
        \end{itemize}
      }
    \end{block}
  \end{column}
\end{columns}
